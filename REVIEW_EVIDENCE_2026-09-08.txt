#include <iostream>
#include <iomanip>
#include <sstream>

#include "D:/Project/Chain/Transaction/Block.h"
#include "D:/Project/Chain/Message/Message.h"

int observed = 0;
void check(const char* name, bool reproduces) {
    std::cout << name << "=" << (reproduces ? "REPRODUCED" : "NOT_REPRODUCED") << '\n';
    observed += reproduces;
}

int main() {
    if (sodium_init() < 0) return 2;
    spdlog::set_level(spdlog::level::off);
    auto wallet = GenerateWallet();
    auto tx = GenerateTx(wallet.public_key, wallet.public_key, 1, 1, wallet);
    if (!VerifyTransaction(tx)) return 3;
    auto altered = tx;
    altered[64] ^= 0x01;
    std::array<uint8_t, 32> recomputed{};
    crypto_generichash(recomputed.data(), 32, altered.data(), 80, nullptr, 0);
    check("modified_amount_accepted", VerifyTransaction(altered) && recomputed != GetTransactionHash(altered));

    auto invalid = tx;
    invalid[112] ^= 0x01;
    if (VerifyTransaction(invalid)) return 4;
    Block block{};
    block.height = 2;
    block.txNum = 1;
    block.merkleRoot = GetTransactionHash(invalid);
    auto header = SerializeBlockHeader(block);
    crypto_generichash(block.hash.data(), 32, header.data(), 80, nullptr, 0);
    std::vector<uint8_t> encoded(112 + 176);
    std::memcpy(encoded.data(), header.data(), 80);
    std::memcpy(encoded.data() + 80, block.hash.data(), 32);
    std::memcpy(encoded.data() + 112, invalid.data(), 176);
    check("invalid_signature_block_accepted", VerifyBlock(encoded));

    auto tx2 = GenerateTx(wallet.public_key, wallet.public_key, 1, 2, wallet);
    std::vector<uint8_t> package(352);
    std::memcpy(package.data(), tx.data(), 176);
    std::memcpy(package.data() + 176, tx2.data(), 176);
    txpool.clear();
    ProcessTxPackage(package);
    check("two_tx_package_keeps_first_only", txpool.size() == 1 && txpool.count(GetTransactionHash(tx2)) == 0);
    txpool.clear();
    ProcessTxPackage(std::vector<uint8_t>(invalid.begin(), invalid.end()));
    check("package_accepts_invalid_signature", txpool.size() == 1);

    auto framed = GenerateNewBlockMessage(encoded);
    uint32_t declared = 0;
    std::memcpy(&declared, framed.data() + 1, 4);
    check("block_frame_length_five_too_large", declared == framed.size() && declared == framed.size() - 5 + 5);
    auto time = GetTime();
    uint64_t timestamp = 1;
    std::memcpy(&timestamp, time.data(), 8);
    check("uninitialized_time_is_zero", timestamp == 0);

    // InitDB uses the process working directory; run only in an isolated work folder.
    InitDB();
    if (!db) return 5;
    GenerateGenesisBlock();
    DBWriteBlockALL(block.hash, encoded);
    DBWriteCurrentBlock(block.hash);
    DBWriteBlockHeight(2);
    auto byHeight = DBReadBlockByHeight("2");
    check("height_lookup_returns_hash_not_block", byHeight.size() == 32 && byHeight != encoded);
    GenerateGenesisBlock();
    check("genesis_resets_existing_head", DBReadBlockHeight() == 1 && DBReadCurrentBlock() != block.hash);
    db.reset();
    std::cout << "reproduced=" << observed << "/8\n";
    return observed == 8 ? 0 : 1;
}
